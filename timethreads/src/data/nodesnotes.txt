Example Queries:

SELECT DISTINCT ?war ?warLabel ?startDate ?endDate
WHERE
{
  ?war wdt:P31/wdt:P279* wd:Q350604
  OPTIONAL{?war wdt:P580 ?startDate .}     # start date
  OPTIONAL{?war wdt:P582 ?endDate . }    # end date
  
  SERVICE wikibase:label {
    bd:serviceParam wikibase:language "[AUTO_LANGUAGE],mul,en".
  } 
}
LIMIT 150

cy.on("mouseover", "edge", (event) => {
  const edge = event.target;
  const position = event.renderedPosition;

  setHoveredEdge({
    text: edge.data("relationship"),
    x: position.x,
    y: position.y,
  });
});

cy.on("mouseout", "edge", () => {
  setHoveredEdge(null);
});

{hoveredEdge && (
    <div
      style={{
        position: "absolute",
        left: hoveredEdge.x + 12,
        top: hoveredEdge.y + 12,
        padding: "8px 12px",
        background: "#1f2937",
        color: "white",
        borderRadius: "6px",
        fontSize: "13px",
        pointerEvents: "none",
        zIndex: 10,
      }}
    >
      {hoveredEdge.text}
    </div>
  )}
