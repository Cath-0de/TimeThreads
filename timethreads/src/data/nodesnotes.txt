cy.on("mouseover", "edge", (event) => {
  const edge = event.target;
  const position = event.renderedPosition;

  setHoveredEdge({
    text: edge.data("relationship"),
    x: position.x,
    y: position.y,
  });
});

cy.on("mouseout", "edge", () => {
  setHoveredEdge(null);
});
